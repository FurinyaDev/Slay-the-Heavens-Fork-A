local function writes(grid)
 local count={refreshes=0,geometry=0,frame=0,sizeReads=0};local seen={}
 local itemSize=grid.GetItemSize
 grid.GetItemSize=function(self)count.sizeReads=count.sizeReads+1;return itemSize(self)end
 local names={'SetAnchoredPosition','SetSizeDelta','SetLocalScale','SetLocalRotation','SetVisible'}
 local function watch(c,frame)
  if seen[c] then return end
  seen[c]=true
  frame=frame or c.name=='stsRef' or c.name=='stsReference' or c.name=='stsCardReference'
  for _,n in ipairs(names)do
   if c[n] then
    local setter=c[n]
    c[n]=function(self,...)
     count.geometry=count.geometry+1
     if frame then count.frame=count.frame+1 end
     return setter(self,...)
    end
   end
  end
  for _,child in ipairs(c:GetChildren())do watch(child,frame)end
 end
 local refresh=grid.RefreshItems
 grid.RefreshItems=function(self,n,fn)
  count.refreshes=count.refreshes+1
  return refresh(self,n,function(cell,index) watch(cell);return fn(cell,index)end)
 end
 return count
end

end

end
